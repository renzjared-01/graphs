## Treap
from dataclasses import dataclass, field
from random import Random

rand = Random()
def random_priority():
    return rand.getrandbits(64)

@dataclass(slots=True)
class Node:
    value: int
    priority: int = field(default_factory=random_priority)
    l: object = field(default=None)
    r: object = field(default=None)

def split(t, v):
    if t is None:
        return None, None, None

    if v == t.value:
        l, r = t.l, t.r
        t.l = None
        t.r = None
        return l, t, r

    if v < t.value:
        l, m, t.l = split(t.l, v)
        return l, m, t
    else:
        t.r, m, r = split(t.r, v)
        return t, m, r

def merge(l, r):
    if l is None:
        return r
    if r is None:
        return l

    if l.priority > r.priority:
        l.r = merge(l.r, r)
        return l
    else:
        r.l = merge(l, r.l)
        return r

def contains(t, v):
    while t is not None:
        if v == t.value:
            return True
        t = t.l if v < t.value else t.r
    return False

def next_larger(t, v):
    res = None
    while t is not None:
        if v < t.value:
            res = t.value
            t = t.l
        else:
            t = t.r
    return res


class OrderedSet:
    def __init__(self):
        self.root = None

    def contains(self, v):
        return contains(self.root, v)

    def insert(self, v):
        l, m, r = split(self.root, v)
        if m is None:
            m = Node(v)
        self.root = merge(merge(l, m), r)

    def delete(self, v):
        l, _, r = split(self.root, v)
        self.root = merge(l, r)

    def next_larger(self, v):
        return next_larger(self.root, v)

newtreap = OrderedSet()

# Advanced DFS
def bridges_and_articulation_points(n, edges):
    adj = [[] for _ in range(n)]
    for idx, (i, j) in enumerate(edges):
        assert 0 <= i < n
        assert 0 <= j < n
        adj[i].append((j, idx))
        adj[j].append((i, idx))

    disc = [-1]*n
    low = [-1]*n

    is_artic = [False]*n
    is_bridge = [False]*len(edges)

    time = 0
    def dfs(i, pidx=-1):
        nonlocal time
        disc[i] = time
        time += 1

        low[i] = disc[i]
        children_count = 0
        for j, idx in adj[i]:
            if idx == pidx:  # this is the parent edge
                continue

            if disc[j] == -1:  # this is a tree edge
                dfs(j, idx)
                low[i] = min(low[i], low[j])

                if low[j] > disc[i]:
                    is_bridge[idx] = True

                if low[j] >= disc[i]:
                    is_artic[i] = True

                children_count += 1

            else:  # this is a back edge
                low[i] = min(low[i], disc[j])

        if pidx == -1:
            is_artic[i] = children_count >= 2

    # run a DFS on all components
    for s in range(n):
        if disc[s] == -1:
            dfs(s)

    bridges = [(i, j) for bridge, (i, j) in zip(is_bridge, edges, strict=True) if bridge]
    artics = [i for i in range(n) if is_artic[i]]

    return bridges, artics
