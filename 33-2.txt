## Treap
from dataclasses import dataclass, field
from random import Random

rand = Random()
def random_priority():
    return rand.getrandbits(64)

@dataclass(slots=True)
class Node:
    value: int
    priority: int = field(default_factory=random_priority)
    l: object = field(default=None)
    r: object = field(default=None)

def split(t, v):
    if t is None:
        return None, None, None

    if v == t.value:
        l, r = t.l, t.r
        t.l = None
        t.r = None
        return l, t, r

    if v < t.value:
        l, m, t.l = split(t.l, v)
        return l, m, t
    else:
        t.r, m, r = split(t.r, v)
        return t, m, r

def merge(l, r):
    if l is None:
        return r
    if r is None:
        return l

    if l.priority > r.priority:
        l.r = merge(l.r, r)
        return l
    else:
        r.l = merge(l, r.l)
        return r

def contains(t, v):
    while t is not None:
        if v == t.value:
            return True
        t = t.l if v < t.value else t.r
    return False

def next_larger(t, v):
    res = None
    while t is not None:
        if v < t.value:
            res = t.value
            t = t.l
        else:
            t = t.r
    return res


class OrderedSet:
    def __init__(self):
        self.root = None

    def contains(self, v):
        return contains(self.root, v)

    def insert(self, v):
        l, m, r = split(self.root, v)
        if m is None:
            m = Node(v)
        self.root = merge(merge(l, m), r)

    def delete(self, v):
        l, _, r = split(self.root, v)
        self.root = merge(l, r)

    def next_larger(self, v):
        return next_larger(self.root, v)

from __future__ import annotations
from dataclasses import dataclass, field

@dataclass(slots=True)
class Node:
    value: object
    h: int = 0
    l: Node | None = field(default=None)
    r: Node | None = field(default=None)

    @staticmethod
    def height(node):
        return -1 if node is None else node.h

    def update_height(self):
        self.h = max(Node.height(self.l), Node.height(self.r)) + 1

    def rotate_left(self):
        y = self.r
        self.r = y.l
        y.l = self
        self.update_height()
        y.update_height()
        return y

    def rotate_right(self):
        y = self.l
        self.l = y.r
        y.r = self
        self.update_height()
        y.update_height()
        return y

    def rebalance(self):
        self.update_height()
        balance = Node.height(self.l) - Node.height(self.r)

        # Left heavy
        if balance > 1:
            if Node.height(self.l.l) < Node.height(self.l.r):
                self.l = self.l.rotate_left()
            return self.rotate_right()

        # Right heavy
        if balance < -1:
            if Node.height(self.r.r) < Node.height(self.r.l):
                self.r = self.r.rotate_right()
            return self.rotate_left()

        return self
    
    def insert(self, value):
        if value < self.value:
            self.l = self.l.insert(value) if self.l else Node(value)
        elif value > self.value:
            self.r = self.r.insert(value) if self.r else Node(value)
        # duplicates ignored
        return self.rebalance()

    def delete(self, value):
        if value < self.value:
            if self.l:
                self.l = self.l.delete(value)
        elif value > self.value:
            if self.r:
                self.r = self.r.delete(value)
        else:  # found node to delete
            if self.l is None:
                return self.r
            elif self.r is None:
                return self.l
            else:
                self.value, self.r = self.r.delete_smallest()
        return self.rebalance()

    def delete_smallest(self):
        if self.l is None:
            return self.value, self.r
        else:
            val, self.l = self.l.delete_smallest()
            return val, self.rebalance()

    def contains(self, value):
        if value == self.value:
            return True
        elif value < self.value:
            return self.l.contains(value) if self.l else False
        else:
            return self.r.contains(value) if self.r else False

    def next_larger(self, value):
        if value < self.value:
            if self.l:
                res = self.l.next_larger(value)
                if res is not None:
                    return res
            return self.value
        else:
            return self.r.next_larger(value) if self.r else None

class OrderedSet:
    def __init__(self):
        self.root = None

    def insert(self, value):
        self.root = self.root.insert(value) if self.root else Node(value)

    def delete(self, value):
        if self.root:
            self.root = self.root.delete(value)

    def contains(self, value):
        return self.root.contains(value) if self.root else False

    def next_larger(self, value):
        return self.root.next_larger(value) if self.root else None


# Advanced DFS
def bridges_and_articulation_points(n, edges):
    adj = [[] for _ in range(n)]
    for idx, (i, j) in enumerate(edges):
        assert 0 <= i < n
        assert 0 <= j < n
        adj[i].append((j, idx))
        adj[j].append((i, idx))

    disc = [-1]*n
    low = [-1]*n

    is_artic = [False]*n
    is_bridge = [False]*len(edges)

    time = 0
    def dfs(i, pidx=-1):
        nonlocal time
        disc[i] = time
        time += 1

        low[i] = disc[i]
        children_count = 0
        for j, idx in adj[i]:
            if idx == pidx:  # this is the parent edge
                continue

            if disc[j] == -1:  # this is a tree edge
                dfs(j, idx)
                low[i] = min(low[i], low[j])

                if low[j] > disc[i]:
                    is_bridge[idx] = True

                if low[j] >= disc[i]:
                    is_artic[i] = True

                children_count += 1

            else:  # this is a back edge
                low[i] = min(low[i], disc[j])

        if pidx == -1:
            is_artic[i] = children_count >= 2

    # run a DFS on all components
    for s in range(n):
        if disc[s] == -1:
            dfs(s)

    bridges = [(i, j) for bridge, (i, j) in zip(is_bridge, edges, strict=True) if bridge]
    artics = [i for i in range(n) if is_artic[i]]

    return bridges, artics

## Kosaraju's
# find strongly connected components
def kosaraju(n, edges):
    adj = [[] for e in range(n)]
    adj_r = [[] for e in range(n)]
    for i, j in edges:
        adj[i].append(j)
        adj_r[j].append(i) #reversed

    visited = [False] * n

    def dfs(i, adj, out):
        visited[i] = True
        for j in adj[i]:
            if not visited[j]:
                dfs(j, adj, out)
        out.append(i)

    stack = []
    for i in range(n):
        if not visited[i]:
            dfs(i, adj, stack)

    visited[:] = [False] * n
    for i in reversed(stack): #decr order
        if not visited[i]:
            scc = []
            dfs(i, adj_r, scc)
            yield scc

#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

#define MAX_N 100 // max vertices
typedef struct Node {
    int val;
    struct Node* next;
} Node;

void add_edge(Node* adj[], int u, int v) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->val = v;
    newNode->next = adj[u];
    adj[u] = newNode;
}

void dfs(int u, Node* adj[], bool visited[], int stack[], int* top, int n) {
    visited[u] = true;
    for (Node* curr = adj[u]; curr != NULL; curr = curr->next) {
        if (!visited[curr->val])
            dfs(curr->val, adj, visited, stack, top, n);
    }
    stack[(*top)++] = u;
}

void dfs_collect(int u, Node* adj[], bool visited[], int scc[], int* scc_size) {
    visited[u] = true;
    scc[(*scc_size)++] = u;
    for (Node* curr = adj[u]; curr != NULL; curr = curr->next) {
        if (!visited[curr->val])
            dfs_collect(curr->val, adj, visited, scc, scc_size);
    }
}

int main() {
    int n = 7;
    int edges[][2] = {
        {1, 2},
        {2, 3},
        {3, 1},
        {0, 4},
        {4, 3},
        {5, 0},
        {0, 6},
        {6, 5}
    };
    int m = sizeof(edges)/sizeof(edges[0]); // # of edge

    Node* adj[MAX_N] = {0};
    Node* adj_r[MAX_N] = {0};
    // Node** adj   = calloc(n, sizeof(Node*));
    // Node** adj_r = calloc(n, sizeof(Node*));

    for (int i = 0; i < m; i++) {
        int u = edges[i][0];
        int v = edges[i][1];
        add_edge(adj, u, v);
        add_edge(adj_r, v, u);  //reversed
    }

    bool visited[MAX_N] = {0};
    int stack[MAX_N];
    // bool* visited = calloc(n, sizeof(bool));
    // int* stack = malloc(n * sizeof(int));
    int top = 0;
    for (int i=0; i< n; i++) {
        if (!visited[i])
            dfs(i, adj, visited, stack, &top, n);
    }

    for (int i=0; i<n; i++)
        visited[i] = false;
    for (int i = top-1; i >= 0; i--) {
        int node = stack[i];
        if (!visited[node]) {
            int scc[MAX_N];
            int scc_size = 0;
            dfs_collect(node, adj_r, visited, scc, &scc_size);

            printf("{ ");
            for (int j = 0; j < scc_size; j++)
                printf("%d ", scc[j]);
            printf("}\n");
        }
    }

    for (int i = 0; i < n; i++) {
        Node* curr = adj[i];
        while (curr) {
            Node* temp = curr;
            curr = curr->next;
            free(temp);
        }
        curr = adj_r[i];
        while (curr) {
            Node* temp = curr;
            curr = curr->next;
            free(temp);
        }
    }
    return 0;
}
