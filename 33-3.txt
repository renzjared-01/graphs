def hierholzer(n, edges):
    adj = [[] for _ in range(n)]
    for idx, (i, j) in enumerate(edges):
        adj[i].append((j, idx))
        adj[j].append((i, idx))

    degree = [0] * n
    for i, j in edges:
        degree[i] += 1
        degree[j] += 1

    # start
    odd = [i for i in range(n) if degree[i] % 2]
    if len(odd) == 0:
        start = next((i for i in range(n) if degree[i] > 0), 0)
    elif len(odd) == 2:
        start = odd[0]
    else:
        raise ValueError('no eulerian path')
        # no eulerian path

    used = [False]*len(edges)
    stack = [start]
    edge_stack = []   # store edge indices parallel to stack
    res = []

    while stack:
        v = stack[-1]

        # skip used edges
        while adj[v] and used[adj[v][-1][1]]:
            adj[v].pop()

        if adj[v]:
            u, idx = adj[v].pop()
            used[idx] = True
            stack.append(u)
            edge_stack.append(idx)
        else:
            stack.pop()
            if edge_stack:
                res.append(edge_stack.pop())
    
    if not all(used):
        raise ValueError('not all edges used')

    return start, res[::-1]

class Node:
    def __init__(self, arr, i, j):
        self.i = i
        self.j = j

        if self.is_leaf:
            self.min = arr[i]
            self.l = None
            self.r = None
        else:
            k = (i + j) // 2
            self.l = Node(arr, i, k)
            self.r = Node(arr, k, j)
            self.combine()
        super().__init__()

    def combine(self):
        self.min = min(self.l.min, self.r.min)

    def range_min(self, i, j):
        if i <= self.i and self.j <= j:
            return self.min
        elif j <= self.i or self.j <= i:
            return float('inf')
        else:
            return min(
                self.l.range_min(i, j),
                self.r.range_min(i, j)
            )

    def assign(self, i, v):
        if i >= self.i and i < self.j:
            if self.is_leaf:
                self.min = v
            else:
                self.l.assign(i, v)
                self.r.assign(i, v)
                self.combine()

    @property
    def is_leaf(self):
        return self.j - self.i == 1


class RangeMin:
    def __init__(self, a):
        a = list(a)
        self.root = Node(a, 0, len(a))
        super().__init__()

    def range_min(self, i, j):
        return self.root.range_min(i, j)

    def __setitem__(self, i, v):
        self.root.assign(i, v)

class HeavyLightDecomposition:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]

        self.parent = [-1] * n
        self.depth = [0] * n
        self.heavy = [-1] * n
        self.head = [0] * n
        self.pos = [0] * n
        self.subsize = [0] * n

        self.cur_pos = 0

    def add_edge(self, u, v):
        self.adj[u].append(v)
        self.adj[v].append(u)

    # First DFS: compute subtree sizes + heavy child
    def dfs(self, v, p):
        self.parent[v] = p
        self.subsize[v] = 1
        max_subsize = 0

        for u in self.adj[v]:
            if u != p:
                self.depth[u] = self.depth[v] + 1
                self.dfs(u, v)

                self.subsize[v] += self.subsize[u]

                if self.subsize[u] > max_subsize:
                    max_subsize = self.subsize[u]
                    self.heavy[v] = u

    # Second DFS: assign head + position
    def decompose(self, v, h):
        self.head[v] = h
        self.pos[v] = self.cur_pos
        self.cur_pos += 1

        # Continue heavy path
        if self.heavy[v] != -1:
            self.decompose(self.heavy[v], h)

        # Decompose light edges
        for u in self.adj[v]:
            if u != self.parent[v] and u != self.heavy[v]:
                self.decompose(u, u)

    def build(self, root=0):
        self.dfs(root, -1)
        self.decompose(root, root)

def query_path(hld, u, v, segtree):
    res = 0  # or identity element

    while hld.head[u] != hld.head[v]:
        if hld.depth[hld.head[u]] < hld.depth[hld.head[v]]:
            u, v = v, u

        head_u = hld.head[u]
        res += segtree.query(hld.pos[head_u], hld.pos[u])
        u = hld.parent[head_u]

    # same head
    if hld.depth[u] > hld.depth[v]:
        u, v = v, u

    res += segtree.query(hld.pos[u], hld.pos[v])
    return res
