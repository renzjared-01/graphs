def hierholzer(n, edges):
    adj = [[] for _ in range(n)]
    for idx, (i, j) in enumerate(edges):
        adj[i].append((j, idx))
        adj[j].append((i, idx))

    degree = [0] * n
    for i, j in edges:
        degree[i] += 1
        degree[j] += 1

    # start
    odd = [i for i in range(n) if degree[i] % 2]
    if len(odd) == 0:
        start = next((i for i in range(n) if degree[i] > 0), 0)
    elif len(odd) == 2:
        start = odd[0]
    else:
        raise ValueError('no eulerian path')
        # no eulerian path

    used = [False]*len(edges)
    stack = [start]
    edge_stack = []   # store edge indices parallel to stack
    res = []

    while stack:
        v = stack[-1]

        # skip used edges
        while adj[v] and used[adj[v][-1][1]]:
            adj[v].pop()

        if adj[v]:
            u, idx = adj[v].pop()
            used[idx] = True
            stack.append(u)
            edge_stack.append(idx)
        else:
            stack.pop()
            if edge_stack:
                res.append(edge_stack.pop())
    
    if not all(used):
        raise ValueError('not all edges used')

    return start, res[::-1]

#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct EdgeNode {
    int to;
    int edge_idx;
    struct EdgeNode* next;
} EdgeNode;

typedef struct {
    EdgeNode** adj;
    int* degree;
    int n;
    int num_edges;
} Graph;

void add_edge(Graph* g, int u, int v, int idx) {
    EdgeNode* node = (EdgeNode*)malloc(sizeof(EdgeNode));
    node->to = v;
    node->edge_idx = idx;
    node->next = g->adj[u];
    g->adj[u] = node;
    g->degree[u]++;
}

void solve_hierholzer(int n, int num_edges, int edge_pairs[][2]) {
    Graph g;
    g.n = n;
    g.num_edges = num_edges;
    g.adj = (EdgeNode**)calloc(n, sizeof(EdgeNode*));
    g.degree = (int*)calloc(n, sizeof(int));

    for (int i = 0; i < num_edges; i++) {
        add_edge(&g, edge_pairs[i][0], edge_pairs[i][1], i);
        add_edge(&g, edge_pairs[i][1], edge_pairs[i][0], i);
    }

    // 1. Find Start Node
    int start = -1;
    int odd_count = 0;
    int odd_nodes[2];
    for (int i = 0; i < n; i++) {
        if (g.degree[i] % 2 != 0) {
            if (odd_count < 2) odd_nodes[odd_count] = i;
            odd_count++;
        }
    }

    if (odd_count == 0) {
        for (int i = 0; i < n; i++) if (g.degree[i] > 0) { start = i; break; }
        if (start == -1) start = 0;
    } else if (odd_count == 2) {
        start = odd_nodes[0];
    } else {
        printf("Error: No Eulerian Path exists (odd degrees: %d)\n", odd_count);
        return;
    }

    // 2. Traversal Data Structures
    bool* used = (bool*)calloc(num_edges, sizeof(bool));
    int* stack = (int*)malloc((num_edges + 1) * sizeof(int));
    int* edge_stack = (int*)malloc(num_edges * sizeof(int));
    int* res = (int*)malloc(num_edges * sizeof(int));
    
    int s_ptr = 0, es_ptr = 0, res_ptr = 0;
    stack[s_ptr++] = start;

    // 3. Main Loop
    while (s_ptr > 0) {
        int v = stack[s_ptr - 1];

        // Clean up used edges at the head of the list
        while (g.adj[v] && used[g.adj[v]->edge_idx]) {
            EdgeNode* tmp = g.adj[v];
            g.adj[v] = g.adj[v]->next;
            free(tmp);
        }

        if (g.adj[v]) {
            // Forward move
            EdgeNode* edge = g.adj[v];
            int u = edge->to;
            int idx = edge->edge_idx;

            used[idx] = true;
            g.adj[v] = g.adj[v]->next; // "Pop" from adj list
            free(edge);

            stack[s_ptr++] = u;
            edge_stack[es_ptr++] = idx;
        } else {
            // Backtrack
            s_ptr--;
            if (es_ptr > 0) {
                res[res_ptr++] = edge_stack[--es_ptr];
            }
        }
    }

    // 4. Validation and Output
    if (res_ptr != num_edges) {
        printf("Error: Not all edges used. Graph might be disconnected.\n");
    } else {
        printf("Start node: %d\nPath (edges): ", start);
        // Note: res is filled in postorder, so printing reverse for result order
        for (int i = res_ptr - 1; i >= 0; i--) {
            printf("%d ", res[i]);
        }
        printf("\n");
    }

    // Cleanup
    free(used); free(stack); free(edge_stack); free(res); free(g.degree);
    for (int i = 0; i < n; i++) {
        EdgeNode* curr = g.adj[i];
        while (curr) { EdgeNode* next = curr->next; free(curr); curr = next; }
    }
    free(g.adj);
}

int main() {
    int edges[7][2] = {{0,1}, {1,2}, {2,0}, {0,4}, {4,3}, {3,0}, {4,5}};
    solve_hierholzer(6, 7, edges);
    return 0;
}

class RootedTree:
    def __init__(self, n, edges, root_idx=0):
        self.n = n
        self.k = n.bit_length()
        self.adj = [[] for _ in range(n)]
        for u, v in edges:
            self.adj[u].append(v)
            self.adj[v].append(u)

        self.jump = [[-1] * n for _ in range(self.k)]
        self.depth = [-1] * n
        self._preprocess(root_idx)

    def _preprocess(self, root_idx):
        stack = [(root_idx, -1, 0)]
        self.depth[root_idx] = 0
        
        while stack:
            u, p, d = stack.pop()
            self.jump[0][u] = p
            self.depth[u] = d
            for v in self.adj[u]:
                if v != p:
                    stack.append((v, u, d + 1))

        for i in range(1, self.k):
            for u in range(self.n):
                mid = self.jump[i-1][u]
                if mid != -1:
                    self.jump[i][u] = self.jump[i-1][mid]

    def lca(self, u, v):
        if self.depth[u] < self.depth[v]:
            u, v = v, u
        
        diff = self.depth[u] - self.depth[v]
        for i in range(self.k):
            if (diff >> i) & 1:
                u = self.jump[i][u]

        if u == v:
            return u

        for i in reversed(range(self.k)):
            if self.jump[i][u] != self.jump[i][v]:
                u = self.jump[i][u]
                v = self.jump[i][v]

        return self.jump[0][u]

typedef struct Node {
    int v;
    struct Node* next;
} Node;

typedef struct {
    int n;
    int k;
    int* depth;
    int** jump;
    Node** adj;
} RootedTree;

// Helper to add edge to adjacency list
void add_edge(Node** adj, int u, int v) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    new_node->v = v;
    new_node->next = adj[u];
    adj[u] = new_node;
}

RootedTree* create_tree(int n, int edges[][2], int num_edges, int root_idx) {
    RootedTree* tree = (RootedTree*)malloc(sizeof(RootedTree));
    tree->n = n;
    
    // Calculate K (log2 of n)
    int k = 0;
    while ((1 << k) <= n) k++;
    tree->k = k;

    tree->adj = (Node**)calloc(n, sizeof(Node*));
    for (int i = 0; i < num_edges; i++) {
        add_edge(tree->adj, edges[i][0], edges[i][1]);
        add_edge(tree->adj, edges[i][1], edges[i][0]);
    }

    tree->depth = (int*)malloc(n * sizeof(int));
    tree->jump = (int**)malloc(k * sizeof(int*));
    for (int i = 0; i < k; i++) {
        tree->jump[i] = (int*)malloc(n * sizeof(int));
        for (int j = 0; j < n; j++) tree->jump[i][j] = -1;
    }

    // Iterative DFS for preprocessing
    int* stack = (int*)malloc(n * sizeof(int));
    int top = -1;
    
    stack[++top] = root_idx;
    tree->depth[root_idx] = 0;
    tree->jump[0][root_idx] = -1;

    // Use a temp array to track visited since it's an undirected graph being treated as rooted
    int* visited = (int*)calloc(n, sizeof(int));
    visited[root_idx] = 1;

    while (top >= 0) {
        int u = stack[top--];
        for (Node* curr = tree->adj[u]; curr != NULL; curr = curr->next) {
            int v = curr->v;
            if (!visited[v]) {
                visited[v] = 1;
                tree->depth[v] = tree->depth[u] + 1;
                tree->jump[0][v] = u;
                stack[++top] = v;
            }
        }
    }

    // Fill binary lifting table
    for (int i = 1; i < k; i++) {
        for (int u = 0; u < n; u++) {
            int mid = tree->jump[i - 1][u];
            if (mid != -1) {
                tree->jump[i][u] = tree->jump[i - 1][mid];
            }
        }
    }

    free(stack);
    free(visited);
    return tree;
}

int lca(RootedTree* tree, int u, int v) {
    if (tree->depth[u] < tree->depth[v]) {
        int temp = u; u = v; v = temp;
    }

    // Lift u to the same depth as v
    int diff = tree->depth[u] - tree->depth[v];
    for (int i = 0; i < tree->k; i++) {
        if ((diff >> i) & 1) {
            u = tree->jump[i][u];
        }
    }

    if (u == v) return u;

    // Lift both simultaneously
    for (int i = tree->k - 1; i >= 0; i--) {
        if (tree->jump[i][u] != tree->jump[i][v]) {
            u = tree->jump[i][u];
            v = tree->jump[i][v];
        }
    }

    return tree->jump[0][u];
}

int main() {
    int n = 7;
    int edges[6][2] = {{0, 1}, {0, 2}, {1, 6}, {5, 1}, {3, 1}, {5, 4}};
    
    RootedTree* tree = create_tree(n, edges, 6, 0);

    printf("%d\n", lca(tree, 6, 3)); // Output: 1
    printf("%d\n", lca(tree, 6, 4)); // Output: 1
    printf("%d\n", lca(tree, 6, 2)); // Output: 0
    printf("%d\n", lca(tree, 1, 4)); // Output: 1

    return 0;
}

class Node:
    def __init__(self, arr, i, j):
        self.i = i
        self.j = j

        if self.is_leaf:
            self.min = arr[i]
            self.l = None
            self.r = None
        else:
            k = (i + j) // 2
            self.l = Node(arr, i, k)
            self.r = Node(arr, k, j)
            self.combine()
        super().__init__()

    def combine(self):
        self.min = min(self.l.min, self.r.min)

    def range_min(self, i, j):
        if i <= self.i and self.j <= j:
            return self.min
        elif j <= self.i or self.j <= i:
            return float('inf')
        else:
            return min(
                self.l.range_min(i, j),
                self.r.range_min(i, j)
            )

    def assign(self, i, v):
        if i >= self.i and i < self.j:
            if self.is_leaf:
                self.min = v
            else:
                self.l.assign(i, v)
                self.r.assign(i, v)
                self.combine()

    @property
    def is_leaf(self):
        return self.j - self.i == 1


class RangeMin:
    def __init__(self, a):
        a = list(a)
        self.root = Node(a, 0, len(a))
        super().__init__()

    def range_min(self, i, j):
        return self.root.range_min(i, j)

    def __setitem__(self, i, v):
        self.root.assign(i, v)

class HeavyLightDecomposition:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]

        self.parent = [-1] * n
        self.depth = [0] * n
        self.heavy = [-1] * n
        self.head = [0] * n
        self.pos = [0] * n
        self.subsize = [0] * n

        self.cur_pos = 0

    def add_edge(self, u, v):
        self.adj[u].append(v)
        self.adj[v].append(u)

    # First DFS: compute subtree sizes + heavy child
    def dfs(self, v, p):
        self.parent[v] = p
        self.subsize[v] = 1
        max_subsize = 0

        for u in self.adj[v]:
            if u != p:
                self.depth[u] = self.depth[v] + 1
                self.dfs(u, v)

                self.subsize[v] += self.subsize[u]

                if self.subsize[u] > max_subsize:
                    max_subsize = self.subsize[u]
                    self.heavy[v] = u

    # Second DFS: assign head + position
    def decompose(self, v, h):
        self.head[v] = h
        self.pos[v] = self.cur_pos
        self.cur_pos += 1

        # Continue heavy path
        if self.heavy[v] != -1:
            self.decompose(self.heavy[v], h)

        # Decompose light edges
        for u in self.adj[v]:
            if u != self.parent[v] and u != self.heavy[v]:
                self.decompose(u, u)

    def build(self, root=0):
        self.dfs(root, -1)
        self.decompose(root, root)

def query_path(hld, u, v, segtree):
    res = 0  # or identity element

    while hld.head[u] != hld.head[v]:
        if hld.depth[hld.head[u]] < hld.depth[hld.head[v]]:
            u, v = v, u

        head_u = hld.head[u]
        res += segtree.query(hld.pos[head_u], hld.pos[u])
        u = hld.parent[head_u]

    # same head
    if hld.depth[u] > hld.depth[v]:
        u, v = v, u

    res += segtree.query(hld.pos[u], hld.pos[v])
    return res
