## Range Min
class FlatRangeMin:
    def __init__(self, data, max_n=None, inf=float('inf')):
        self.inf = inf
        n = len(data)
        if max_n is None:
            max_n = n
        
        self.N = 1
        while self.N < max_n:
            self.N *= 2
            
        # Index 0 is unused; 1 is root; N to 2N-1 are leaves
        self.tree = [inf] * (2 * self.N)
        
        for i in range(n):
            self.tree[self.N + i] = data[i]
            
        for i in range(self.N - 1, 0, -1):
            self.tree[i] = min(self.tree[2 * i], self.tree[2 * i + 1])

    def update(self, i, val):
        idx = self.N + i
        self.tree[idx] = val
        while idx > 1:
            idx //= 2  # Move to parent
            self.tree[idx] = min(self.tree[2 * idx], self.tree[2 * idx + 1])

    def query(self, l, r):
        res = self.inf
        l += self.N
        r += self.N
        while l < r:
            if l % 2 == 1: # If l is a right child, it's the only part of its parent we need
                res = min(res, self.tree[l])
                l += 1
            if r % 2 == 1: # If r is a right child, r-1 is the left child we need
                r -= 1
                res = min(res, self.tree[r])
            l //= 2
            r //= 2
        return res


class LazyNode(Node):
    __slots__ = 'i', 'j', 'm', 'l', 'r', 'tag'
    def __init__(self, a, i, j):
        self.tag = 0  # Pending value to be added to children
        super().__init__(a, i, j)

    def _apply(self, v):
        """Apply an update to this node and mark it for children."""
        self.m += v
        self.tag += v

    def push(self):
        """Push the pending tag down to children."""
        if self.tag != 0 and not self.is_leaf:
            self.l._apply(self.tag)
            self.r._apply(self.tag)
            self.tag = 0

    def range_add(self, i, j, v):
        # 1. Full Overlap: Update this node and stop
        if i <= self.i and self.j <= j:
            self._apply(v)
            return
        
        # 2. No Overlap: Do nothing
        if j <= self.i or self.j <= i:
            return

        # 3. Partial Overlap: Push pending tags down, then recurse
        self.push()
        self.l.range_add(i, j, v)
        self.r.range_add(i, j, v)
        self.combine()

    def range_min(self, i, j, inf):
        if i <= self.i and self.j <= j:
            return self.m
        if j <= self.i or self.j <= i:
            return inf
        
        self.push() # Crucial: push tags before querying children
        return min(self.l.range_min(i, j, inf), self.r.range_min(i, j, inf))

class HeavyLightDecomposition:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]

        self.parent = [-1] * n
        self.depth = [0] * n
        self.heavy = [-1] * n
        self.head = [0] * n
        self.pos = [0] * n
        self.subsize = [0] * n

        self.cur_pos = 0

    def add_edge(self, u, v):
        self.adj[u].append(v)
        self.adj[v].append(u)

    # First DFS: compute subtree sizes + heavy child
    def dfs(self, v, p):
        self.parent[v] = p
        self.subsize[v] = 1
        max_subsize = 0

        for u in self.adj[v]:
            if u != p:
                self.depth[u] = self.depth[v] + 1
                self.dfs(u, v)

                self.subsize[v] += self.subsize[u]

                if self.subsize[u] > max_subsize:
                    max_subsize = self.subsize[u]
                    self.heavy[v] = u

    # Second DFS: assign head + position
    def decompose(self, v, h):
        self.head[v] = h
        self.pos[v] = self.cur_pos
        self.cur_pos += 1

        # Continue heavy path
        if self.heavy[v] != -1:
            self.decompose(self.heavy[v], h)

        # Decompose light edges
        for u in self.adj[v]:
            if u != self.parent[v] and u != self.heavy[v]:
                self.decompose(u, u)

    def build(self, root=0):
        self.dfs(root, -1)
        self.decompose(root, root)

def query_path(hld, u, v, segtree):
    res = 0  # or identity element

    while hld.head[u] != hld.head[v]:
        if hld.depth[hld.head[u]] < hld.depth[hld.head[v]]:
            u, v = v, u

        head_u = hld.head[u]
        res += segtree.query(hld.pos[head_u], hld.pos[u])
        u = hld.parent[head_u]

    # same head
    if hld.depth[u] > hld.depth[v]:
        u, v = v, u

    res += segtree.query(hld.pos[u], hld.pos[v])
    return res

from dataclasses import dataclass, field
from typing import Callable

from sample_graphs import T_ud_edgelist as G1_edgelist

@dataclass(slots=True)
class TreeNode:
    id: int
    val: int
    children: "list[TreeNode]" = field(default_factory=list)
    depth: int = -1
    weight: int = 1
    p: "TreeNode | None" = None
    top: "TreeNode | None" = None

    def is_leaf(self):
        return not self.children


def edgelist_to_adj(n: int, el: list[tuple[int, int]]) -> dict[int, list[int]]:
    adj = {i: [] for i in range(n)}
    for i, j in el:
        adj[i].append(j)
        adj[j].append(i)
    return adj


class HeavyLightDecomp:
    def __init__(self, adj: dict[int, list[int]], values: list[int], f: Callable[[int, int], int], f_ident: int, r: int = 0):
        self._adj = dict(adj)
        self._f = f
        self._f_ident = f_ident

        self._root, self._nodes = self._get_rooted_tree(values, r)
        self._compute_tops()
        self._flat, self._node_id_to_idx = self._get_heavy_paths()

        self._values = [self._nodes[i].val for i in self._flat]
        self._range_query = RangeQuery(self._values, f, f_ident)

    def path_query(self, u: int, v: int):
        u_node = self._nodes[u]
        v_node = self._nodes[v]
        ans = self._f_ident

        while True:
            u_idx = self._node_id_to_idx[u_node.id]
            v_idx = self._node_id_to_idx[v_node.id]
            if u_node.top.depth == v_node.top.depth:
                res = self._range_query.get(min(u_idx, v_idx), max(u_idx, v_idx) + 1)
                ans = self._f(ans, res)
                break
            elif u_node.top.depth > v_node.top.depth:
                top_idx = self._node_id_to_idx[u_node.top.id]
                res = self._range_query.get(min(u_idx, top_idx), max(u_idx, top_idx) + 1)
                ans = self._f(ans, res)
                u_node = u_node.top.p
            else:
                top_idx = self._node_id_to_idx[v_node.top.id]
                res = self._range_query.get(min(v_idx, top_idx), max(v_idx, top_idx) + 1)
                ans = self._f(ans, res)
                v_node = v_node.top.p
        return ans

    def set(self, i: int, val: int):
        self._nodes[i].val = val
        self._values[self._node_id_to_idx[i]] = val
        self._range_query.set(i, val)


    def _get_rooted_tree(self, vals: list[int], r: int):
        n = len(self._adj)
        nodes = [None] * n

        def _dfs(i, p):
            node = TreeNode(i, vals[i])
            nodes[i] = node
            node.p = nodes[p]
            node.depth = node.p.depth + 1
            for j in self._adj[i]:
                if j != p:
                    child = _dfs(j, i)
                    node.children.append(child)
                    node.weight += child.weight
            heavy_child_idx, _ = max(
                (tup for tup in enumerate(node.children)),
                key=lambda t: (t[1].weight, -t[1].id),
                default=(-1, None)
            )
            if heavy_child_idx != -1:
                node.children[0], node.children[heavy_child_idx] = \
                    node.children[heavy_child_idx], node.children[0]
            return node

        root = _dfs(r, r)
        return root, nodes

    def _compute_tops(self):
        assert self._root is not None
        # compute topmost node of heavy path the node belongs to
        def _dfs(node):
            if node.p is node:
                node.top = node
            elif node.p.children[0] is not node:
                node.top = node
            else:
                node.top = node.p.top
            for ch in node.children:
                _dfs(ch)
        _dfs(self._root)

    def _get_heavy_paths(self):
        # assumes heavy child is at children[0]
        flat = []
        node_id_to_idx = {}

        def _dfs(node):
            node_id_to_idx[node.id] = len(flat)
            flat.append(node.id)
            for ch in node.children:
                _dfs(ch)
        _dfs(self._root)
        return (flat, node_id_to_idx)

def edgelist_to_adj(n, el):
    adj = {i: [] for i in range(n)}
    for i, j in el:
        adj[i].append(j)
        adj[j].append(i)
    return adj


def dfs_size(adj, i, p, done):
    sz = 1
    for j in adj[i]:
        if j != p and j not in done:
            sz += dfs_size(adj, j, i, done)
    return sz


def dfs_centroid(adj, i, done):
    n = dfs_size(adj, i, None, done)
    centroid = None

    def _dfs(i, p):
        nonlocal centroid
        wi = 1
        ok_centroid = True

        for j in adj[i]:
            if j != p and j not in done:
                wj = _dfs(j, i)
                wi += wj
                if wj * 2 > n:
                    # size of subtree under j is too large
                    ok_centroid = False
        if wi * 2 < n:  # size of all subtrees of i is too small
            ok_centroid = False
        if ok_centroid:
            centroid = i
        return wi

    _dfs(i, None)
    return centroid


def centroid_dnc(adj, i):
    done = set()

    def _centroid_dnc(i):
        centroid = dfs_centroid(adj, i, done)
        # for calculations that require the centroid, do it here
        done.add(centroid)
        print(centroid, end=' ')
        for j in adj[centroid]:
            if j not in done:
                _centroid_dnc(j)

    _centroid_dnc(i)
